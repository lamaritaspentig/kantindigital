Kode Google Apps Script (Secure & Private)

Salin kode ini ke Apps Script Google Sheets Anda. Data transaksi siswa dikunci aman dan foto bukti bayar otomatis disimpan ke Google Drive.

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();

  // ===================================================
  // PENGATURAN MANUAL KANTIN (EDIT DI SINI)
  // ===================================================
  var MANUAL_CONFIG = {
    qrisUrl: "https://api.qrserver.com/v1/create-qr-code/?size=450x450&data=00020101021126580014ID.GO.QRIS.WWW01189360091400000000000215KANTINSEKOLAH0153033605802ID5914IRMA%20JAYA5303360",
    adminPin: "3333", // PIN Kasir Rahasia
    cashierWa: "081234567890"
  };
  // ===================================================

  // 1. Ambil/Inisialisasi Stok dari Sheet "STOK"
  var sheetStok = getOrCreateSheet(ss, "STOK");
  var stokData = sheetStok.getDataRange().getDisplayValues();
  var menuItems = [];

  if (stokData.length <= 1) {
    var defaultItems = [
      { id: 'v1000', name: 'Voucher Belanja', price: 1000, stock: 0, icon: '🎟️' },
      { id: 'v2000', name: 'Voucher Belanja', price: 2000, stock: 0, icon: '🎟️' },
      { id: 'v3000', name: 'Voucher Belanja', price: 3000, stock: 0, icon: '🎟️' },
      { id: 'v5000', name: 'Voucher Belanja', price: 5000, stock: 0, icon: '🎟️' },
      { id: 'v10000', name: 'Voucher Belanja', price: 10000, stock: 0, icon: '🎟️' }
    ];
    sheetStok.clearContents();
    sheetStok.appendRow(["ID", "NAMA_VOUCHER", "HARGA", "STOK", "ICON"]);
    defaultItems.forEach(function(item) {
      sheetStok.appendRow([item.id, item.name, item.price, item.stock, item.icon]);
    });
    menuItems = defaultItems;
  } else {
    for (var i = 1; i < stokData.length; i++) {
      if (stokData[i][0]) {
        menuItems.push({
          id: String(stokData[i][0]),
          name: String(stokData[i][1]),
          price: Number(stokData[i][2]),
          stock: Number(stokData[i][3]),
          icon: String(stokData[i][4] || '🎟️')
        });
      }
    }
  }

  // 2. Keamanan: Hanya kirim data transaksi jika PIN Kasir valid
  var reqPin = String(e && e.parameter && e.parameter.pin || '').trim();
  var orders = [];

  if (reqPin === MANUAL_CONFIG.adminPin) {
    var sheetOrders = getOrCreateSheet(ss, "TRANSAKSI");
    var ordersData = sheetOrders.getDataRange().getDisplayValues();
    var startIdx = Math.max(1, ordersData.length - 200);

    for (var j = ordersData.length - 1; j >= startIdx; j--) {
      if (ordersData[j][0]) {
        var parsedVouchers = [];
        try {
          parsedVouchers = JSON.parse(ordersData[j][11] || '[]');
        } catch (err) {
          parsedVouchers = [];
        }

        orders.push({
          id: String(ordersData[j][0]),
          date: String(ordersData[j][1]),
          buyerName: String(ordersData[j][2]),
          buyerNisn: String(ordersData[j][3]),
          buyerClass: String(ordersData[j][4]),
          buyerPhone: String(ordersData[j][5]),
          itemsSummary: String(ordersData[j][6]),
          totalAmount: Number(ordersData[j][7]),
          paymentStatus: String(ordersData[j][8] || 'Belum Bayar'),
          redeemStatus: String(ordersData[j][9] || 'Belum Ditukar'),
          paymentProof: String(ordersData[j][10] || ''),
          vouchers: parsedVouchers
        });
      }
    }
  }

  // PIN Kasir TIDAK PERNAH dikirim dalam respon JSON!
  return ContentService.createTextOutput(JSON.stringify({
    qrisUrl: MANUAL_CONFIG.qrisUrl,
    cashierWa: MANUAL_CONFIG.cashierWa,
    menuItems: menuItems,
    orders: orders
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(10000);

    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();

    // Action: Update Transaksi Keseluruhan
    if (data.action === "update_orders") {
      var sheetOrders = getOrCreateSheet(ss, "TRANSAKSI");
      clearSheetDataRows(sheetOrders);
      if (data.orders && Array.isArray(data.orders)) {
        data.orders.forEach(function(o) {
          sheetOrders.appendRow([
            o.id, o.date, o.buyerName, o.buyerNisn, o.buyerClass,
            o.buyerPhone, o.itemsSummary, o.totalAmount,
            o.paymentStatus, o.redeemStatus, o.paymentProof || '', JSON.stringify(o.vouchers || [])
          ]);
        });
      }
      return ContentService.createTextOutput("Orders Updated");
    }

    // Action: Update Stok Keseluruhan
    if (data.action === "update_stock") {
      var sheetStok = getOrCreateSheet(ss, "STOK");
      clearSheetDataRows(sheetStok);
      if (data.menuItems && Array.isArray(data.menuItems)) {
        data.menuItems.forEach(function(m) {
          sheetStok.appendRow([m.id, m.name, m.price, m.stock, m.icon || '🎟️']);
        });
      }
      return ContentService.createTextOutput("Stock Updated");
    }

    // Process Gambar Bukti Pembayaran: Simpan ke Google Drive
    var proofUrl = "";
    if (data.paymentProof && data.paymentProof.indexOf("data:image") === 0) {
      try {
        var folderName = "BUKTI_PEMBAYARAN_KANTIN";
        var folders = DriveApp.getFoldersByName(folderName);
        var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);

        var base64Data = data.paymentProof.split(",")[1];
        var blob = Utilities.newBlob(Utilities.base64Decode(base64Data), "image/jpeg", "BUKTI_" + data.buyerName + "_" + Date.now() + ".jpg");
        var file = folder.createFile(blob);
        file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
        proofUrl = "https://lh3.googleusercontent.com/d/" + file.getId();
      } catch (errDrive) {
        proofUrl = data.paymentProof;
      }
    } else {
      proofUrl = data.paymentProof || '';
    }

    // Default Action: Tambah Transaksi Baru & Kurangi Stok Otomatis
    var sheetDefault = getOrCreateSheet(ss, "TRANSAKSI");
    sheetDefault.appendRow([
      data.id, data.date, data.buyerName, data.buyerNisn, data.buyerClass, 
      data.buyerPhone, data.itemsSummary, data.totalAmount, 
      data.paymentStatus, data.redeemStatus, proofUrl, JSON.stringify(data.vouchers || [])
    ]);

    // Update stok jika ada
    if (data.vouchers && Array.isArray(data.vouchers)) {
      var sheetStok = getOrCreateSheet(ss, "STOK");
      var stokData = sheetStok.getDataRange().getDisplayValues();
      
      var usage = {};
      data.vouchers.forEach(function(v) {
        var nominal = v.nominal;
        var itemId = 'v' + nominal;
        usage[itemId] = (usage[itemId] || 0) + 1;
      });

      for (var i = 1; i < stokData.length; i++) {
        var currentId = String(stokData[i][0]);
        if (usage[currentId]) {
          var currentStock = Number(stokData[i][3]);
          var newStock = Math.max(0, currentStock - usage[currentId]);
          sheetStok.getRange(i + 1, 4).setValue(newStock);
        }
      }
    }

    return ContentService.createTextOutput("Order Saved & Stock Deducted");
  } catch(err) {
    return ContentService.createTextOutput("Error: " + err.toString());
  } finally {
    lock.releaseLock();
  }
}

function clearSheetDataRows(sheet) {
  var lastRow = sheet.getLastRow();
  if (lastRow > 1) {
    sheet.getRange(2, 1, lastRow - 1, sheet.getLastColumn()).clearContent();
  }
}

function getOrCreateSheet(ss, name) {
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    if (name === "TRANSAKSI") {
      sheet.appendRow(["ORDER_ID", "TANGGAL", "NAMA_SISWA", "NISN", "KELAS", "NO_HP", "ITEMS", "TOTAL", "STATUS_BAYAR", "STATUS_TUKAR", "BUKTI_PEMBAYARAN", "VOUCHERS_JSON"]);
    } else if (name === "STOK") {
      sheet.appendRow(["ID", "NAMA_VOUCHER", "HARGA", "STOK", "ICON"]);
    }
  }
  return sheet;
}
