function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetConfig = getOrCreateSheet(ss, "CONFIG");
  var sheetOrders = getOrCreateSheet(ss, "TRANSAKSI");
  
  var config = {};
  var configData = sheetConfig.getDataRange().getValues();
  for (var i = 1; i < configData.length; i++) {
    if (configData[i][0]) config[configData[i][0]] = configData[i][1];
  }
  
  var orders = [];
  var ordersData = sheetOrders.getDataRange().getValues();
  for (var j = 1; j < ordersData.length; j++) {
    if (ordersData[j][0]) {
      orders.push({
        id: ordersData[j][0],
        date: ordersData[j][1],
        buyerName: ordersData[j][2],
        buyerNisn: ordersData[j][3],
        buyerClass: ordersData[j][4],
        buyerPhone: ordersData[j][5],
        itemsSummary: ordersData[j][6],
        totalAmount: Number(ordersData[j][7]),
        paymentStatus: ordersData[j][8],
        redeemStatus: ordersData[j][9],
        vouchers: JSON.parse(ordersData[j][10] || '[]')
      });
    }
  }
  
  return ContentService.createTextOutput(JSON.stringify({
    qrisUrl: config.qrisUrl || '',
    adminPin: String(config.adminPin || '3333'),
    menuItems: JSON.parse(config.menuItems || '[]'),
    orders: orders
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    if (data.action === "update_config") {
      var sheetConfig = getOrCreateSheet(ss, "CONFIG");
      if (data.qrisUrl !== undefined) setConfigValue(sheetConfig, "qrisUrl", data.qrisUrl);
      if (data.adminPin !== undefined) setConfigValue(sheetConfig, "adminPin", String(data.adminPin));
      if (data.menuItems !== undefined) setConfigValue(sheetConfig, "menuItems", JSON.stringify(data.menuItems));
      return ContentService.createTextOutput("Config Saved");
    } 

    if (data.action === "update_orders") {
      var sheetOrders = getOrCreateSheet(ss, "TRANSAKSI");
      sheetOrders.clearContents();
      sheetOrders.appendRow(["ORDER_ID", "TANGGAL", "NAMA_SISWA", "NISN", "KELAS", "NO_HP", "ITEMS", "TOTAL", "STATUS_BAYAR", "STATUS_TUKAR", "VOUCHERS_JSON"]);
      if (data.orders && Array.isArray(data.orders)) {
        data.orders.forEach(function(o) {
          sheetOrders.appendRow([
            o.id, o.date, o.buyerName, o.buyerNisn, o.buyerClass,
            o.buyerPhone, o.itemsSummary, o.totalAmount,
            o.paymentStatus, o.redeemStatus, JSON.stringify(o.vouchers || [])
          ]);
        });
      }
      return ContentService.createTextOutput("Orders Updated");
    }
    
    // Default: New Single Transaction
    var sheetOrdersDefault = getOrCreateSheet(ss, "TRANSAKSI");
    sheetOrdersDefault.appendRow([
      data.id, data.date, data.buyerName, data.buyerNisn, data.buyerClass, 
      data.buyerPhone, data.itemsSummary, data.totalAmount, 
      data.paymentStatus, data.redeemStatus, JSON.stringify(data.vouchers || [])
    ]);
    
    return ContentService.createTextOutput("Order Saved");
  } catch(err) {
    return ContentService.createTextOutput("Error: " + err.toString());
  }
}

function getOrCreateSheet(ss, name) {
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    if (name === "CONFIG") {
      sheet.appendRow(["KEY", "VALUE"]);
    } else if (name === "TRANSAKSI") {
      sheet.appendRow(["ORDER_ID", "TANGGAL", "NAMA_SISWA", "NISN", "KELAS", "NO_HP", "ITEMS", "TOTAL", "STATUS_BAYAR", "STATUS_TUKAR", "VOUCHERS_JSON"]);
    }
  }
  return sheet;
}

function setConfigValue(sheet, key, value) {
  var data = sheet.getDataRange().getValues();
  for (var i = 1; i < data.length; i++) {
    if (data[i][0] === key) {
      sheet.getRange(i + 1, 2).setValue(value);
      return;
    }
  }
  sheet.appendRow([key, value]);
}
